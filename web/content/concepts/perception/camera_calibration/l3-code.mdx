## The fit

<CodeWalk title="Bundle adjustment over intrinsics and per-view poses">

```python title="calibrate.py"
import numpy as np
from scipy.optimize import least_squares

def calibrate(views, board, f0=800.0, cx0=None, cy0=None):
    """views: list of (N,2) detected corner arrays. board: (N,3) metric points.

    The per-view poses are nuisance parameters optimised jointly with the
    intrinsics — which is why the residual can be driven to noise level even
    when the intrinsics are badly wrong: the poses absorb the error.
    """
    n = len(views)
    x0 = np.concatenate([[f0, f0, cx0, cy0]] + [np.r_[0, 0, 0, 0, 0, 0.5]] * n)

    def residuals(x):
        K4 = x[:4]
        return np.concatenate([
            (project(K4, x[4 + 6 * i : 7 + 6 * i], x[7 + 6 * i : 10 + 6 * i], board)
             - views[i]).ravel()
            for i in range(n)
        ])

    sol = least_squares(residuals, x0, method="trf")
    rms = np.sqrt((sol.fun ** 2).mean())
    return sol.x[:4], sol, rms
```

## The check the residual cannot give you

<CodeWalk title="Conditioning from the Jacobian at the solution">

```python title="conditioning.py"
import numpy as np

def parameter_conditioning(sol, n_intrinsics=4):
    """A flat direction shows up as a tiny singular value of the Jacobian —
    available for free at the solution, and invisible in the residual.

    In the fronto-parallel case two different initialisations converged to
    6000.1 and 6000.0 with 0.184 px RMS. The residual said everything was fine;
    the conditioning would have said the focal direction was unconstrained.
    """
    J = sol.jac
    s = np.linalg.svd(J, compute_uv=False)
    print(f"  condition number: {s[0] / s[-1]:.3e}")

    # Which parameter combination is least constrained?
    _, _, Vt = np.linalg.svd(J, full_matrices=False)
    weakest = Vt[-1][:n_intrinsics]
    names = ["fx", "fy", "cx", "cy"]
    print("  weakest direction, intrinsic components:")
    for nm, w in zip(names, weakest):
        print(f"    {nm}: {w:+.3f}")


def parameter_std(sol, n_obs):
    """Approximate standard errors: sigma^2 * (J^T J)^{-1}, with sigma^2 the
    residual variance. Wide intervals on fx are the alarm."""
    dof = max(n_obs - len(sol.x), 1)
    s2 = (sol.fun ** 2).sum() / dof
    cov = s2 * np.linalg.pinv(sol.jac.T @ sol.jac)
    return np.sqrt(np.diag(cov))[:4]
```

</CodeWalk>

<KeyIdea>
Two lines of SVD convert a silent failure into a printed number. The reason
almost nobody does it is that the toolbox already prints a reassuring scalar, and
a second, less familiar number that contradicts it feels redundant. It is the
opposite — the residual and the conditioning answer different questions, and only
the second one is about the parameters.
</KeyIdea>

## Capture protocol, enforced in code

<CodeWalk title="Reject a capture set that cannot determine the parameters">

```python title="protocol.py"
import numpy as np

def check_capture(poses, min_tilt_deg=20.0, min_views_tilted=6):
    """Refuse to calibrate from geometry that cannot identify f.

    Frontal views leave f and Z coupled: N views give N scale measurements and
    N+1 unknowns, so no number of them is enough. Tilt is the fix and 10 deg
    already brings the error to ~1%.
    """
    tilts = np.array([np.rad2deg(np.arccos(np.clip(rod(r)[2, 2], -1, 1)))
                      for r, _ in poses])
    n_tilted = int((tilts > min_tilt_deg).sum())
    depths = np.array([t[2] for _, t in poses])

    ok = n_tilted >= min_views_tilted and depths.ptp() > 0.3 * depths.mean()
    print(f"  views tilted > {min_tilt_deg}°: {n_tilted}/{len(poses)}")
    print(f"  depth range: {depths.min():.2f}–{depths.max():.2f} m")
    if not ok:
        raise ValueError("capture geometry cannot identify the focal length")
    return True


def check_handeye_motions(motions, min_axis_angle_deg=30.0):
    """Two motions with near-parallel rotation axes are formally sufficient and
    numerically useless: 0.15 deg of noise gave 3.02 deg of error at 5 deg
    separation, against 0.41 deg at 90. The solver will not complain."""
    axes = [logm(R) / max(np.linalg.norm(logm(R)), 1e-9) for R, _ in motions]
    best = max(np.rad2deg(np.arccos(abs(np.clip(a @ b, -1, 1))))
               for i, a in enumerate(axes) for b in axes[i + 1:])
    print(f"  largest angle between motion axes: {best:.1f}°")
    if best < min_axis_angle_deg:
        raise ValueError("hand-eye motions are near-coaxial; add a different rotation")
    return True
```

## The validation that actually closes the loop

<CodeWalk title="Measure a metric quantity the calibration never saw">

```python title="validate.py"
def touch_test(robot, perceive_fn, targets, ruler_readings):
    """Command the robot to touch known points through the full vision pipeline
    and measure the miss physically.

    This is the only check that leaves the reprojection metric, and therefore
    the only one a degenerate calibration cannot pass. Sample across the
    workspace, not one point: hand-eye rotation error appears as a bias that
    VARIES with position, so a single point can be fitted by a wrong transform.
    """
    errs = []
    for tgt, measured in zip(targets, ruler_readings):
        p = perceive_fn(tgt)
        robot.move_to(p)
        errs.append(measured)
        print(f"  target {tgt}: miss {measured * 1000:.1f} mm")

    import numpy as np
    e = np.array(errs)
    print(f"  mean {e.mean()*1000:.1f} mm   max {e.max()*1000:.1f} mm")
    # A constant offset points at hand-eye translation; a position-dependent
    # one points at hand-eye rotation or at the focal length.
    return e
```

</CodeWalk>

<Warning>
The pattern in the last line is the diagnostic. A miss that is the same
everywhere is a translation offset. A miss that grows with distance from the
camera is a focal-length or hand-eye rotation error. A miss that grows towards
the image edges is distortion. Recording *where* in the workspace each
measurement was taken costs nothing and turns one number into three
distinguishable hypotheses.
</Warning>
